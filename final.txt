def get_user_db(email):
    safe_email = email.replace("@", "_").replace(".", "_")
    path = f"data/{safe_email}.db"
    # ✅ fresh connection every time
    conn = sqlite3.connect(path, check_same_thread=False)
    conn.execute("PRAGMA journal_mode=WAL;")
    c = conn.cursor()
    # Chat table
    c.execute("""
    CREATE TABLE IF NOT EXISTS chat_history (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        chat_name TEXT,
        sender TEXT,
        message TEXT
    )
    """)
    # Memory table
    c.execute("""
    CREATE TABLE IF NOT EXISTS memory (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        note TEXT
    )
    """)
    # Conversation summary table
    c.execute("""
    CREATE TABLE IF NOT EXISTS summaries (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        summary TEXT
    )
    """)
    # Pinned chats table
    c.execute("""
    CREATE TABLE IF NOT EXISTS pinned_chats (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        chat_name TEXT
    )
    """)
    conn.commit()
    return conn, c

def load_chat(email, chat_name):
    conn, c = get_user_db(email)
    c.execute("""
    SELECT sender, message 
    FROM chat_history 
    WHERE chat_name=? 
    ORDER BY id ASC
    """, (chat_name,))
    data = c.fetchall()
    decrypted_data = []
    for sender, message in data:
        try:
            message = decrypt_text(message)
        except Exception as e:
            print("Decrypt Error:", e)
        decrypted_data.append((sender,message))
    data = decrypted_data
    conn.close()
    return data

# ================= SMART MEMORY =================
def smart_save_memory(email, message):
    existing = load_memory(email)
    # duplicate protection
    for mem in existing:
        if message.lower().strip() == mem.lower().strip():
            return False
    # 🔒 Memory limit
    if len(existing) >= 200:
        return False
    save_memory(email, message)
    return True

def extract_memory_with_ai(user_message):

Message:
{user_message}
Rules:
- Only keep useful personal information
- Ignore temporary chat

- One memory per line
- Examples:

User studies engineering
Memories:
"""

            "http://localhost:11434/api/generate",
            json={
                "model": MODEL_NAME,
                "prompt": prompt,
                "stream": False,
                "temperature": 0.3
            },
            timeout=60
        )
        data = response.json()
        memory_text = data.get("response", "").strip()
        if not memory_text:
            return []
        memories = [
            m.strip()
            for m in memory_text.split("\n")
            if m.strip()
        ]
        return memories
    except Exception as e:
        print("Memory Extract Error:", e)
        return []

def speak_safe(text):
    if len(text) > 1000:
        text = text[:1000]
    def run():
        try:
            import pyttsx3
            engine = pyttsx3.init()
            engine.setProperty('rate', 170)
            engine.say(text)
            engine.runAndWait()
            engine.stop()
        except Exception as e:
            print("Voice Error:", e)
    threading.Thread(target=run, daemon=True).start()
def listen():
    r = sr.Recognizer()
    try:
        with sr.Microphone() as source:
            st.info("🎤 Listening...")
            # Background noise adjust
            r.adjust_for_ambient_noise(source, duration=1)
            # Listen with limits
            audio = r.listen(source, timeout=7, phrase_time_limit=12)
            # Try English

def send_otp_email(receiver_email, otp):
    try:
        sender_email = os.getenv("EMAIL")
        sender_password = os.getenv("PASSWORD")
        
        subject = "Your OTP Code"
        message = f"Subject: {subject}\n\nYour OTP is: {otp}"
        server = smtplib.SMTP("smtp.gmail.com", 587)
        server.starttls()
        if not sender_email or not sender_password:
            raise Exception("EMAIL or PASSWORD missing in .env")
        server.login(sender_email, sender_password)
        server.sendmail(sender_email,receiver_email,message)
        server.quit()
        return True
    except Exception as e:
        print("OTP EMAIL ERROR:", e)
        return False
